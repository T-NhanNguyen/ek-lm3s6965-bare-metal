cmake_minimum_required(VERSION 3.20)
project(ftp_external_consumer C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT BSP_ROOT OR NOT LWIP_ROOT)
    message(FATAL_ERROR "Pass BSP_ROOT (only src/include needed) and offline LWIP_ROOT")
endif()
set(LM3S6965_COMPILE_FLAGS -mcpu=cortex-m3 -mthumb -mfloat-abi=soft -ffunction-sections -fdata-sections -fno-common -Wall -Wextra -Werror)
set(LM3S6965_OPTIMIZATION_FLAGS -Os)
add_subdirectory("${BSP_ROOT}/include" headers)
add_subdirectory("${BSP_ROOT}/src" reusable)
# The interface and all configuration paths are owned by this caller.
add_library(caller_lwip INTERFACE)
target_include_directories(caller_lwip INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/config" "${LWIP_ROOT}/src/include")
target_compile_definitions(caller_lwip INTERFACE CALLER_LWIP_CONTRACT=1)
lm3s6965_add_ftp_tcp_library(caller_ftp_tcp caller_lwip)
file(GLOB stack_sources CONFIGURE_DEPENDS "${LWIP_ROOT}/src/core/*.c" "${LWIP_ROOT}/src/core/ipv4/*.c")
add_library(caller_stack STATIC ${stack_sources} "${LWIP_ROOT}/src/netif/ethernet.c")
target_link_libraries(caller_stack PUBLIC caller_lwip)
target_compile_options(caller_stack PRIVATE ${LM3S6965_COMPILE_FLAGS} -Os)
add_executable(consumer consumer.c)
# No manually added include directories or definitions here: prove transitivity.
target_link_libraries(consumer PRIVATE caller_ftp_tcp lm3s6965_ram_file caller_stack)
target_compile_options(consumer PRIVATE ${LM3S6965_COMPILE_FLAGS} -Os)
target_link_options(consumer PRIVATE -mcpu=cortex-m3 -mthumb -mfloat-abi=soft -nostartfiles --specs=nano.specs --specs=nosys.specs -Wl,-e,consumer_entry -Wl,--gc-sections -Wl,-Map=consumer.map)
