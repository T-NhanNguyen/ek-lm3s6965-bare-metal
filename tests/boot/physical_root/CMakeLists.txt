cmake_minimum_required(VERSION 3.20)
project(physical_root_fixture LANGUAGES C ASM)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CMAKE_C_STANDARD 11)
include("${REPO}/cmake/boot_crypto.cmake")
add_library(verifier STATIC "${REPO}/src/boot_verify.c" "${REPO}/src/boot_format.c")
target_include_directories(verifier PUBLIC "${REPO}/include")
target_link_libraries(verifier PUBLIC boot_crypto)
add_executable(root_behavior root_behavior.c)
target_link_libraries(root_behavior PRIVATE verifier)
target_include_directories(root_behavior PRIVATE "${REPO}/tests/boot")
if(CMAKE_CROSSCOMPILING)
    # Actual production target registration, NOT a rewritten adapter on ARM.
    add_subdirectory("${REPO}" bsp EXCLUDE_FROM_ALL)
    add_executable(adapter_closure arm_closure.c)
    # Launch/select consumers introduce boot_verify_stored after extraction.
    # Place the real verifier dependency after those consumers explicitly.
    target_link_libraries(lm3s6965_boot_launch PUBLIC verifier)
    target_link_libraries(adapter_closure PRIVATE lm3s6965_boot_target_io verifier)
    target_compile_options(adapter_closure PRIVATE -mcpu=cortex-m3 -mthumb -mfloat-abi=soft -Os -fno-short-enums)
    target_link_options(adapter_closure PRIVATE -mcpu=cortex-m3 -mthumb -mfloat-abi=soft -nostartfiles --specs=nosys.specs
        "-T${CMAKE_CURRENT_LIST_DIR}/synthetic.ld" -Wl,--gc-sections,-e,fixture_entry "-Wl,-Map=${CMAKE_BINARY_DIR}/closure.map")
    target_compile_options(verifier PRIVATE -mcpu=cortex-m3 -mthumb -mfloat-abi=soft -Os -fno-short-enums)
else()
    if(NOT ADAPTER_COPY)
        message(FATAL_ERROR "Safe generated ADAPTER_COPY required; never compile raw MMIO into host fixture")
    endif()
    add_executable(adapter_host "${ADAPTER_COPY}" adapter_host.c "${REPO}/src/boot_flash.c" "${REPO}/src/boot_store.c" "${REPO}/src/boot_format.c")
    target_include_directories(adapter_host PRIVATE "${REPO}/include" "${CMAKE_CURRENT_LIST_DIR}")
    foreach(t verifier root_behavior adapter_host boot_crypto)
        target_compile_options(${t} PRIVATE -O1 -g -UNDEBUG -fno-short-enums -fsanitize=address,undefined -fno-omit-frame-pointer)
    endforeach()
    foreach(t root_behavior adapter_host)
        target_link_options(${t} PRIVATE -fsanitize=address,undefined)
    endforeach()
    target_compile_options(adapter_host PRIVATE -Wall -Wextra -Werror)
endif()
