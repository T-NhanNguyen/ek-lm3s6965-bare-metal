cmake_minimum_required(VERSION 3.20)
project(boot_fixture_tools LANGUAGES C)
# Native, reusable, out-of-tree entry point; never included by the ARM root.
if(CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "Boot fixture tools require a native host compiler")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(LM3S6965_BOOT_PHYSICAL_PRODUCTION "Require explicit selected PUBLIC root" ON)
if(NOT LM3S6965_BOOT_PHYSICAL_PRODUCTION)
    message(FATAL_ERROR "Fixture tools require an explicit selected PUBLIC root; no demo fallback")
endif()
include(../../cmake/boot_crypto.cmake)
add_library(boot_fixture_metadata STATIC fixture_metadata.c
    ../../src/boot_verify.c ../../src/boot_format.c)
target_include_directories(boot_fixture_metadata PUBLIC . ../../include)
target_link_libraries(boot_fixture_metadata PUBLIC boot_crypto)
add_executable(boot_fixture_metadata_tool metadata_main.c)
target_link_libraries(boot_fixture_metadata_tool PRIVATE boot_fixture_metadata)
option(BOOT_FIXTURE_TOOL_TESTS "Build metadata codec regression executable" OFF)
if(BOOT_FIXTURE_TOOL_TESTS)
    enable_testing()
    add_executable(boot_fixture_metadata_roundtrip ../../tests/boot/test_fixture_metadata.c)
    target_link_libraries(boot_fixture_metadata_roundtrip PRIVATE boot_fixture_metadata)
    add_test(NAME boot_fixture_metadata_roundtrip COMMAND boot_fixture_metadata_roundtrip)
    target_compile_options(boot_fixture_metadata_roundtrip PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
foreach(target IN ITEMS boot_fixture_metadata boot_fixture_metadata_tool)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wpedantic
        -Wconversion -Wsign-conversion -Wshadow -Wstrict-prototypes -Wmissing-prototypes)
endforeach()
